    <script>
        function calculate() {
            const getValue = id => parseFloat(document.getElementById(id).value) || 0;
            const shoe = getValue('shoeWeight');
            const box = getValue('boxWeight');
            const extra = getValue('extraWeight');
            const l = getValue('length');
            const w = getValue('width');
            const h = getValue('height');
            const divisor = getValue('divisor') || 6000;
            const basePrice = getValue('basePrice');
            const addPrice = getValue('addPrice');

            const pairWeightG = shoe * 2;
            const actualWeightKg = (pairWeightG + box + extra) / 1000;
            const volWeightKg = (l * w * h) / divisor;
            const billWeightKg = Math.max(actualWeightKg, volWeightKg);

            let totalCost = 0;
            if (billWeightKg > 0) {
                totalCost = basePrice;
                const remainingWeight = billWeightKg - 0.1;
                if (remainingWeight > 0) {
                    const extraBlocks = Math.ceil((remainingWeight / 0.1) - 1e-6);
                    totalCost += extraBlocks * addPrice;
                }
            }

            document.getElementById('pairWeight').textContent = pairWeightG + ' г';
            document.getElementById('actualWeight').textContent = actualWeightKg.toFixed(3) + ' кг';
            document.getElementById('volWeight').textContent = volWeightKg.toFixed(3) + ' кг';
            document.getElementById('billWeight').textContent = billWeightKg.toFixed(3) + ' кг';
            document.getElementById('totalCost').textContent = '¥ ' + totalCost.toFixed(2);
        }
        document.addEventListener('DOMContentLoaded', calculate);
    </script>
